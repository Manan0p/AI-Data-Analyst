import duckdb
import plotly.express as px

# 1. High performance in-memory scan
con = duckdb.connect()
df = con.execute("""
    SELECT 
        order_id, order_date, region, carrier, order_value, shipping_cost,
        (margin / order_value) * 100 AS margin_pct,
        CASE WHEN carrier = 'FastFreight' AND shipping_cost > 150 THEN 'Anomaly' ELSE 'Normal' END as flag
    FROM read_csv_auto('Q3_Sales_Report.csv')
    WHERE order_date BETWEEN '2024-08-01' AND '2024-08-31'
""").df()

# 2. Compute bivariate distribution & carrier cost variance
fig = px.scatter(
    df, x='order_value', y='shipping_cost',
    color='carrier', symbol='flag',
    hover_data=['order_id', 'margin_pct']
)
fig.show()
